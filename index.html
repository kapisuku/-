function doGet(e) { if (e.parameter.action === 'getData') { var ss = SpreadsheetApp.getActiveSpreadsheet(); var nameSheet = ss.getSheetByName("รายชื่อ"); var logSheet = ss.getSheetByName("เข้าออกเวร"); // ดึงรายชื่อทั้งหมดจากชีท "รายชื่อ" คอลัมน์ B var nameData = nameSheet.getRange(2, 2, nameSheet.getLastRow() - 1, 1).getValues(); var allNames = nameData.map(function(r) { return r[0]; }).filter(function(n) { return n !== ""; }); // ดึงข้อมูลการลงเวลาเพื่อตรวจสอบสถานะ var logData = []; if (logSheet.getLastRow() > 1) { logData = logSheet.getRange(2, 1, logSheet.getLastRow() - 1, 3).getValues(); } // สร้าง Object เก็บสถานะล่าสุดของทุกคน (เริ่มต้นให้เป็น false หรือออฟไลน์ทั้งหมด) var statusMap = {}; allNames.forEach(function(n) { statusMap[n] = false; }); // เช็คประวัติจากบนลงล่าง การกระทำล่าสุดจะเขียนทับสถานะเดิม for (var i = 0; i < logData.length; i++) { var n = logData[i][0]; // คอลัมน์ A (ชื่อ) var timeIn = logData[i][1]; // คอลัมน์ B (เข้า) var timeOut = logData[i][2]; // คอลัมน์ C (ออก) if (n && statusMap.hasOwnProperty(n)) { if (timeIn !== "" && timeOut === "") { statusMap[n] = true; // มีเวลาเข้า แต่ยังไม่มีเวลาออก = เข้าเวรอยู่ (จุดเขียว) } else { statusMap[n] = false; // ออกเวรแล้ว (จุดเทา) } } } // จัดเตรียมข้อมูลส่งกลับให้หน้าเว็บ var result = allNames.map(function(n) { return { name: n, isOnline: statusMap[n] }; }); return ContentService.createTextOutput(JSON.stringify({data: result})) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("เข้าออกเวร"); var data = JSON.parse(e.postData.contents); var name = data.name; var action = data.action; var timestamp = new Date(); if (action === 'เข้าเวร') { var lastRow = sheet.getLastRow(); sheet.getRange(lastRow + 1, 1).setValue(name); sheet.getRange(lastRow + 1, 2).setValue(timestamp); return ContentService.createTextOutput(JSON.stringify({status: "success", msg: "บันทึกเข้าเวรสำเร็จ!"})) .setMimeType(ContentService.MimeType.JSON); } else if (action === 'ออกเวร') { var dataRange = sheet.getRange(1, 1, sheet.getLastRow(), 3).getValues(); for (var i = dataRange.length - 1; i >= 1; i--) { if (dataRange[i][0] === name && dataRange[i][1] !== "" && dataRange[i][2] === "") { sheet.getRange(i + 1, 3).setValue(timestamp); return ContentService.createTextOutput(JSON.stringify({status: "success", msg: "บันทึกออกเวรสำเร็จ!"})) .setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({status: "error", msg: "ไม่พบประวัติการเข้าเวรที่ค้างอยู่ หรือกดออกเวรไปแล้ว"})) .setMimeType(ContentService.MimeType.JSON); } }
